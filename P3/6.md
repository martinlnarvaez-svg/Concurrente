Existe una comisión de 50 alumnos que deben realizar tareas de a pares, las cuales son corregidas por un JTP. Cuando los alumnos llegan, forman una fila. Una vez que están todos en fila, el JTP les asigna un número de grupo a cada uno. Para ello, suponga que existe una función AsignarNroGrupo() que retorna un número “aleatorio” del 1 al 25. Cuando un alumno ha recibido su número de grupo, comienza a realizar su tarea. Al terminarla, el alumno le avisa al JTP y espera por su nota. Cuando los dos alumnos del grupo completaron la tarea, el JTP les asigna un puntaje (el primer grupo en terminar tendrá como nota 25, el segundo 24, y así sucesivamente hasta el último que tendrá nota 1).<br> Nota: el JTP no guarda el número de grupo que le asigna a cada alumno.
~~~
process Alumno[id: 0..49]{
	int equipo,nota;
	
	admin.llegueAlumno(id,equipo);
	// hacer tarea
	admin.termine(equipo,nota);
}

monitor admin{
	int cantidad = 0;
	int equipos[50];
	cond vcBarrera;
	cond comenzarTarea;
	int cantEquipo[25];
	cond equipoTermino[25];
	Cola cola;
	Cola terminados;
	cond termine;
	int notaEquipo[25];
	int puntaje = 25;
	
	procedure llegueAlumno(in int id; out int e){
		cola.push(id);
		cantidad++;
		if(cantidad == 50){
			signal(vcBarrera);
		}
		wait(comenzarTarea);
		e = equipos[id];
	}
	
	procedure termine(in int e; out int nota){
		termiandos.push(e);
		signal(termine);
		wait(equipoTermino[e-1]);
		nota = notaEquipo[e-1];
	}
	
	procedure barrera(){
		wait(vcBarrera);
	}
	
	procedure asignarNumero(in int e){
		equipos[cola.pop] = e;
		signal(comenzarTarea);
	}
	
	procedure recibirTarea(){
		int equipoAux;
		
		if(terminados.isEmty()) wait(termine);
		equipoAux = terminados.pop();
		cantEquipo[equipoAux-1]++;
		if(cantEquipo[equipoAux-1] == 2){
			notaEquipo[equipoAux-1] = puntaje;
			puntaje--;
			signal_all(equipoTermine);
		}
	}
}

procedure JTP{
	int equipo;
	
	admin.barrera();
	for (int i=0; i<50; i++){
		equipo = asignarEquipo();
		admin.asignarNumero(equipo);
	}
	for (int i=0; i<50; i++){
		admin.recibirTarea();
	}
}
