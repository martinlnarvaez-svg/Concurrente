Analice el problema y defina qué procesos, recursos y monitores serán necesarios/convenientes, además de las posibles sincronizaciones requeridas para resolver el problema. Luego, resuelva considerando las siguientes situaciones: <br>
a) Implemente una solución suponiendo que no importa el orden de uso. Existe una función Fotocopiar() que simula el uso de la fotocopiadora. <br>b) Modifique la solución de (a) para el caso en que se deba respetar el orden de llegada. <br>c) Modifique la solución de (b) para el caso en que se deba dar prioridad de acuerdo con la edad de cada persona (cuando la fotocopiadora está libre, la debe usar la persona de mayor edad entre las que estén esperando para usarla). <br>d) Modifique la solución de (a) para el caso en que se deba respetar estrictamente el orden dado por el identificador del proceso (la persona X no puede usar la fotocopiadora hasta que no haya terminado de usarla la persona X-1). <br>e) Modifique la solución de (b) para el caso en que además haya un Empleado que le indica a cada persona cuándo debe usar la fotocopiadora. <br>f) Modificar la solución (e) para el caso en que sean 10 fotocopiadoras. El empleado le indica a la persona qué fotocopiadora usar y cuándo hacerlo.<br>
se necesitaran los procesos de las personas y un monitor que simule un semaforo de acceso a la fotocopiadora<br>

monitor fotocopiadora{
	int fotocopiadoraAsignada[N];
	cond llegue;
	cond seguir[N];
	cond fotocopiadoraLiberada;
	int fotocopiadorasLibres = 10;
	boolean fotocopiadoras[10] = (*,true);
	cola c;
	
	procedure acceder(int id; out int f){
		c.push(id);
		signal(llegue);
		wait(seguir[id]);
		f = fotocopiadoraAsignada[id];
	}
	
	procedure atender(){
		boolean ok = false;
		int i = 0;
		int foto;
	
		if(c.isEmpty) wait(llegue);
		if(fotocopiadorasLibres == 0) wait(fotocopiadoraLiberada);
		while(not ok){
			if(fotocopiadoras[i]){
				ok = true;
				fotocopiadoras[i] = false;
				foto = i;
				fotocopiadotasLibres--;
			}
			else i++;
		}
		int aux = c.pop();
		fotocopiadoraAsignada[aux] = i;
		signal(seguir[aux]);
	}
	
	procedure salir(int f){
		fotocopiadorasLibres++;
		fotocopiadoras[f] = true;
		signal(fotocopiadoraLiberada);
	}
}
