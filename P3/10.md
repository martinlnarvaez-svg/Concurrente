En un parque hay un juego para ser usado por N personas de a una a la vez y de acuerdo al orden en que llegan para solicitar su uso. Además, hay un empleado encargado de desinfectar el juego durante 10 minutos antes de que una persona lo use. Cada persona, al llegar, espera hasta que el empleado le avisa que puede usar el juego, lo usa por un tiempo y luego lo devuelve. <br>Nota: suponga que la persona tiene una función Usar_juego que simula el uso del juego; y el empleado tiene una función Desinfectar_Juego que simula su trabajo. Todos los procesos deben terminar su ejecución.
~~~

process persona[id: 0..N-1]{
	
	admin.solicitarUso();
	Usar_juego();
	admin.salir();
}

process empleado{
	for i=0; i<N; i++{
		admin.atender();
		Desifenctar_juego(); //limpia durante 10 minutos
	}
}

monitor admin{
	cond llegaPersona;
	cond paso;
	cond personaSale;
	cola c;
	
	procedure solicitarUso(){
		c.push(1); // encolo un valor para tener la guarda del if del procedure atender
		signal(llegaPersona);
		wait(paso);
	}
	
	procedure salir(){
		signal(personaSale);
	}
	
	procedure atender(){
		if(c.isEmpty()) wait(llegaPersona);
		c.pop();
		signal(paso);
		wait(personaSale);
	}
}
