Existen N vehículos que deben pasar por un puente de acuerdo con el orden de llegada. Considere que el puente no soporta más de 50000 kg y que cada vehículo cuenta con su propio peso (ningún vehículo supera el peso soportado por el puente).
~~~

process vehiculo[id: 0..N-1]{
	int peso = ...;
	
	puente.pasar(id,peso);
	// pasando
	puente.salir(peso);
}

monitor puente{
	cond autoDormido;
	cola c;
	int pesoRestante = 50000;
	int pesos[N];
	
	procedure pasar(int id, int peso){
		if(not c.isEmpty() or pesoRestante - peso < 0){
			c.push(id);
			pesos[id] = peso;
			wait(autoDormido);
		}
		else pesoRestante -= peso;
	}
	
	procedure salir(int peso){
		int idAux;
		
		pesoRestante += peso;
		while(not c.isEmpty() and pesoRestante - pesos[c.top()] > 0){
			idAux = c.pop();
			pesoRestante -= pesos[idAux];
			signal(autoDormido);
		}
	}

~~~
La función c.top() mira el primer valor de la cola sin sacarlo, osea deja la cola igual y obtiene el primer valor.
