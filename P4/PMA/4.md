Simular la atención en un locutorio con 10 cabinas telefónicas, el cual tiene un empleado que se encarga de atender a 
N clientes. Al llegar, cada cliente espera hasta que el empleado le indique a qué cabina ir, la usa y luego se dirige 
al empleado para pagarle. El empleado atiende a los clientes en el orden en que hacen los pedidos. A cada cliente se le 
entrega un ticket factura por la operación.  
a) Implemente una solución para el problema descrito. 
b) Modifique la solución implementada para que el empleado dé prioridad a los que terminaron de usar la cabina sobre 
los que están esperando para usarla. 
Nota: maximizar la concurrencia; suponga que hay una función Cobrar() llamada por el empleado que simula que el empleado 
le cobra al cliente. 

a)
	chan requisitoCabina(int);
	chan dandoCabina[N](int);
	chan requisitoPago(int);
	chan registroPago[N](Texto);
	
	Process Cliente[id: 0..N-1]{
	    int numCabina;
	    Texto recibo;
	    send requisitoCabina(id);
	    receive dandoCabina[id](numCabina);
	    usarCabina(numCabina);
	    send requisitoPago(id);
	    receive registroPago[id](recibo);
	}
	
	Process Empleado{
	    boolean cabinaLibre[10] = [*,true];
	    int idCli, cabLibre;
	    while(true){
	        if(not empty(requisitoCabina)) -> 
	            receive requisitoCabina(idCli);
	            cabLibre = buscarCabinaLibre(cabinaLibre);
	            cabinaLibre(cabLibre) = false;
	            send dandoCabina[idCli](cabLibre);
	        [] (not Empty(requisitoPago)) ->
	            receive requisitoPago(idCli);
	            Texto recibo = Cobrar();
	            send registroPago[idCli](recibo);
	    }
	}
