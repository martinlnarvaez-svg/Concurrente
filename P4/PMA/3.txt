Se debe modelar el funcionamiento de una casa de comida rápida, en la cual trabajan 2 cocineros y 3 vendedores, y que 
debe atender a C clientes. El modelado debe considerar que:
- Cada cliente realiza un pedido y luego espera a que se lo entreguen. 
- Los pedidos que hacen los clientes son tomados por cualquiera de los vendedores y se lo pasan a los 
cocineros para que realicen el plato. Cuando no hay pedidos para atender, los vendedores aprovechan para reponer un pack 
de bebidas de la heladera (tardan entre 1 y 3 minutos para hacer esto). 
- Repetidamente cada cocinero toma un pedido pendiente dejado por los vendedores, lo cocina y se lo entrega directamente 
al cliente correspondiente. 
Nota: maximizar la concurrencia.

chan pedidos(Texto);
chan recibirPedido[C-1](Texto);
chan retornarPedido[2](int,Texto);
chan pedidosEnCocina(int,Texto);

Process Cocinero[id: 0..1]{
    Texto pedido, pedidoCocinado;
    int idCli;
    while(true){
        receive pedidosEnCocina(idCli,pedido);
        pedidoCocinado = Cocinar(pedido);
        send recibirPedido[idCli](pedidoCocinado);
    }
}

Process Coordinador{
    int idVen,idCli;
    Texto pedido;
    while(true){
        receive recibirPedido(idVen);
        if(empty(pedidos)) send retornarPedido[idVen](-1,"VACIO")
        else{
            receive pedidos(idCli,pedido);
            send retornarPedido[idVen](idCli,pedido);
        }
    }
}

Process Vendedor[id: 0..2]{
    int idCli,idCoc,idVen;
    Texto pedido;
    while(true){
        send recibirPedido(id);
        receive retornarPedido(idCli,pedido);
        if(idCli = -1) delay 60 + random(0..120)
        else send pedidosEnCocina(idCli,pedido);
    }

}

Process Cliente[id: 0..C-1]{
    Texto pedido = generarPedido();
    send pedidos(id,pedido);
    receive recibirPedido[id](pedido);
}
