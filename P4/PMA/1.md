c.
Se podria resolver sin un proceso intermedio pero esto generaria una demora innecesaria ya que al momento que los procesos
Empleado tengan que leer el canal de reportes lo podrian leer como lleno y que solo 1 tome la tarea, por lo que se trabarian
en el siguiente receive o romperian el programa
La mejor solucion es la siguiente:

chan cola(txt,int)
chan respuestas[N](txt)
chan pedido(int);
chan siguiente[2](txt,int);

Process Cliente[id: 0..N-1]{
    txt res, C
    C = ClienteLLega()
    send cola(C,id)
    receive respuestas[id](res)
}

Process Empleado[id: 0..1]{
    txt preg, res
    int idC
    while(true){
        send pedido(id);
        receive siguiente[id](preg,idC)
        if(res = "VACIO") delay(15 mins)
        else{
            res ResolverConsulta(preg)
            send respuestas[idC,res]
        }
    }
}
