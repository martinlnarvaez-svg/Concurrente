En un laboratorio de genética veterinaria hay 3 empleados. El primero de ellos
continuamente prepara las muestras de ADN; cada vez que termina, se la envía al segundo
empleado y vuelve a su trabajo. El segundo empleado toma cada muestra de ADN
preparada, arma el set de análisis que se deben realizar con ella y espera el resultado para
archivarlo. Por último, el tercer empleado se encarga de realizar el análisis y devolverle el
resultado al segundo empleado.

	Process Preparador{
	    while(true){
	        Texto muestraADN = preparar();
	        Admin!muestraPreparada(muestraADN);
	    }
	}
	
	Process ArmadorArchivador{
	    Cola archivos;
	    Texto muestraADN,analisis,set;
	    while(true){
	        Admin!pedido();
	        Admin?entregaPedido(muestraADN);
	        set = armarSet(muestraADN);
	        Analizador!muestraADNArmado(set);
	        Analizador?ADNAnalizado(analisis);
	        archivos.push(analisis);
	    }
	}
	
	Process Analizador{
	    Texto set, analisis;
	    while(true){
	        ArmadorArchivador?muestraADNArmado(set);
	        analisis = analizar(set);
	        ArmadorArchivador!ADNAnalizado(analisis);
	    }
	}
	
	Process Admin{
	    cola muestras;
	    Texto muestra;
	    do Preparador?muestraPreparada(muestra) -> 
	        muestras.push(muestra);
	    [] (not empty(muestras)); ArmadorArchivador?pedido() -> 
	        ArmadorArchivador!entregaPedido(muestras.pop);
	    od
	}
