Simular la atención en una Terminal de Micros que posee 3 puestos para hisopar a 150 pasajeros. En cada puesto hay una Enfermera que atiende a los pasajeros de acuerdo con el orden de llegada al mismo. Cuando llega un pasajero, se dirige al Recepcionista, quien le indica qué puesto es el que tiene menos gente esperando. Luego se dirige al puesto y espera a que la enfermera correspondiente lo llame para hisoparlo. Finalmente, se retira. <br>a) Implemente una solución considerando los procesos Pasajeros, Enfermera y Recepcionista.<br>b) Modifique la solución anterior para que sólo haya procesos Pasajeros y Enfermera, siendo los pasajeros quienes determinan por su cuenta qué puesto tiene menos personas esperando. <br>Nota: suponga que existe una función Hisopar() que simula la atención del pasajero por parte de la enfermera correspondiente.<br>
a)
~~~

sem mutexColaRecepcion = 1;
cola colaRecepcion;
sem llegue = 0;
sem avisoRecepcionista[150] = (*,0);
int puestoAsignado[150];
cola colaP1;
sem mutexColaP1 = 1;
cola colaP2;
sem mutexColaP2 = 1;
cola colaP3;
sem mutexColaP3 = 1;
sem avisoEnfermera = (*,0);
sem avisoEnfermeraRecepcionista[3] = (*,0);
int cantidades[3];
sem mutexCantidades = 1;
int restantes = 150;
sem mutexRestantes = 1;

Process pasajeros[id: 0..149]{
	P(mutexColaRecepcion);
		colaRecepcion.push(id);
	V(mutexColaRecepcion);
	V(llegue);
	P(avisoRecepcionista[id]);
	if(puestoAsignado[id] = 0){
		P(mutexColaP1);
			colaP1.push(id);
		V(mutexColaP1);
	}
	else if(puestoAsignado[id] = 1){
		P(mutexColaP2);
			colaP2.push(id);
		V(mutexColaP2);
	}
	else {
		P(mutexColaP3);
			colaP3.push(id);
		V(mutexColaP3);
	}
	P(avisoEnfermera[id]);
	// hisponado y retirandose
}

Process enfermera[id: 1..3]{
	int idAux;

	P(mutexRestantes);
	while(restantes <> 0){
		V(mutexRestantes);
		P(avisoEnfermeraRecepcionista[id]);
		P(mutexRestantes);
		if(restantes <> 0){
			V(mutexRestantes);
			if(id == 1){
				P(mutexColaP1);
					idAux = colaP1.pop();
				V(mutexColaP1);
				P(mutexCantidades);
					cantidades[0]--;
				V(mutexCantidades);
			}
			else if(id == 2){
				P(mutexColaP2);
					idAux = colaP2.pop();
				V(mutexColaP2);
				P(mutexCantidades);
					cantidades[1]--;
				V(mutexCantidades);
			}
			else {
				P(mutexColaP3);
					idAux = colaP3.pop();
				V(mutexColaP3);
				P(mutexCantidades);
					cantidades[2]--;
				V(mutexCantidades);
			}
			V(avisoEnfermera[idAux]);
			Hisopar(idAux);
			P(mutexRestantes);
			restantes--;
			if(restantes == 0){
				for (int k=0; k<3; k++) V(avisoEnfermeraRecepcionista[k]);
			}
		}
	}
	V(mutexRestantes);
}

Process recepcionista{
	int idAux, puestoMin, cantMin = 999;
	
	for int j=0; i<150; i++{
		P(llegue);
		P(mutexColaRecepcion);
			idAux = colaRecepcion.pop();
		V(mutexColaRecepcion);
		
		P(mutexCantidades);
			for int i=0; i<3; i++{
				if(cantMin < cantidades[i]){
					cantMin = cantidades[i];
					puestoMin = i;
				}
			}
			cantidades[puestoMin]++;
		V(mutexCantidades);
		V(avisoEnfermeraRecepcionista[puestoMin]);
		cantMin = 999;
		
		puestoAsignado[idAux] = puestoMin;
		V(avisoRecepcionista[idAux]);
	}
}
