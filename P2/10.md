A una cerealera van T camiones a descargarse trigo y M camiones a descargar maíz. Sólo hay lugar para que 7 camiones a la vez descarguen, pero no pueden ser más de 5 del mismo tipo de cereal. <br>a) Implemente una solución que use un proceso extra que actúe como coordinador entre los camiones. El coordinador debe atender a los camiones según el orden de llegada. Además, debe retirarse cuando todos los camiones han descargado. <br>b) Implemente una solución que no use procesos adicionales (sólo camiones). No importa el orden de llegada para descargar. Nota: maximice la concurrencia.<br>
a)
~~~

sem capacidad = 7;
sem capacidadTrigo = 5;
sem capacidadMaiz = 5;
sem aviso = 0;
cola c;
sem mutex = 1;
sem avisoTrigo[T] = (*,0);
sem avisoMaiz[M] = (*,0);

Process camionTrigo[id: 0..T-1]{
	P(mutex);
		trigo.push(id,"trigo");
	V(mutex);
	V(aviso);
	P(avisoTrigo[id]);
		descargarMaiz();
		V(capacidadTrigo);
		V(capacidad);
}

Process camionMaiz[id: 0..M-1]{
	P(mutex);
		maiz.push(id,"maiz");
	V(mutex);
	V(aviso);
	P(avisoMaiz[id]);
		descargoTrigo();
		V(capacidadMaiz);
		V(capacidad);
}

Process coordinador{
	int id;
	String tipo;
	
	while(true){
		P(aviso);
		P(mutex);
			c.pop(id,tipo);
		V(mutex);
		if(tipo = "maiz"){
			P(capacidadMaiz);
			P(capacidad);
			V(avisoMaiz[id]);
		}
		else{
			P(capacidadTrigo);
			P(capacidad);
			V(avisoTrigo[id]);
		}
	}
}

~~~
