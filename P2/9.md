Resolver el funcionamiento en una fábrica de ventanas con 7 empleados (4 carpinteros, 1 vidriero y 2 armadores) que trabajan de la siguiente manera: <br>• Los carpinteros continuamente hacen marcos (cada marco es armado por un único carpintero) y los dejan en un depósito con capacidad de almacenar 30 marcos. <br>• El vidriero continuamente hace vidrios y los deja en otro depósito con capacidad para 50 vidrios. <br>• Los armadores continuamente toman un marco y un vidrio (en ese orden) de los depósitos correspondientes y arman la ventana (cada ventana es armada por un único armador).

~~~

sem mutexMarcos = 1;
sem mutexVidrio = 1;
sem capacidadMarcos = 30;
sem capacidadVidrios = 50;
cola marcos;
cola vidrios;
sem avisoMarco = 0;
sem avisoVidrio = 0;
sem hayMarco = 0;
sem hayVidrio = 0;

Process carpintero[id: 0..3]{
	marco m;
	
	while(true){
		hacerMarco(m);
		P(capacidadMarcos);
		P(mutexMarcos);
			marcos.push(m);
		V(mutexMarcos);
		V(avisoMarco);
	}
}

Process Vidriero{
	vidrio v;
	
	while(true){
		hacerVidrio(v);
		P(capacidaVidrio);
		P(mutexVidrio);
			vidrios.push(v);
		V(mutexVidrio);
		V(avisoVidrio);
	}
}

Process Armador[id: 0..1]{
	vidrio v;
	marco m;
	 
	while(true){
		P(avisoMarco);
		P(mutexMarco);
			m = marcos.pop();
		V(mutexMarco);
		V(capacidadMarco);
		
		P(avisoVidrio);
		P(mutexVidrio);
			v = vidrios.pop();
		V(mutexVidrio);
		V(capadidadVidrio);
		
		armarVentana(v,m);
	}
}
