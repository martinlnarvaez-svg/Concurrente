Se requiere modelar un puente de un único sentido que soporta hasta 5 unidades de peso.
El peso de los vehículos depende del tipo: cada auto pesa 1 unidad, cada camioneta pesa 2
unidades y cada camión 3 unidades. Suponga que hay una cantidad innumerable de
vehículos (A autos, B camionetas y C camiones). Analice el problema y defina qué tareas,
recursos y sincronizaciones serán necesarios/convenientes para resolverlo.
a. Realice la solución suponiendo que todos los vehículos tienen la misma prioridad.
b. Modifique la solución para que tengan mayor prioridad los camiones que el resto de los
vehículos.

a)

task Admin is
    entry pasoCamion;
    entry pasoAuto;
    entry pasoCamioneta;
    entry salir(peso: IN integer);
end Admin;

task type Auto is
    autosV: array(1..N) of Auto;
task body Auto is
begin
    Admin.pasoAuto;
    Admin.salir(1);
end Auto;

task type Camioneta is
    camionetasV: array(1..N) of Camioneta;
task body Camioneta is
begin
    Admin.pasoCamioneta;
    Admin.salir(2);
end Camioneta;

task type Camion is
    camionesV: array(1..N) of Camion;
task body Camion is
begin
    Admin.pasoCamion;
    Admin.salir(3);
end Camion;

task body Admin is
    peso: integer:= 0;
begin
    Loop
        Select
            When (peso < 4) => accept pasoAuto do 
                peso:= peso + 1;
            end pasoAuto
        or
            When (peso < 3) => accept pasoCamioneta do
                peso:= peso + 2
            end pasoCamioneta
        or
            When (peso < 2) => accept pasoCamion do
                peso:= peso + 3
            end pasoCamion
        or
            accept salir(pesoSale: IN integer) do
                peso:= peso - pesoSale
            end salir;
        end select;
    end Loop;
end Admin;
