Resolver el siguiente problema con PMS. En la estación de trenes hay una terminal de SUBE que debe ser usada por P personas de acuerdo con el orden de llegada. Cuando la persona accede a la terminal, la usa y luego se retira para dejar al siguiente. <br>Nota: cada Persona usa sólo una vez la terminal.
~~~

process persona[id: 0..P-1]{
	
	admin.llegue(id);
	admin.paso();
	usarTerminal();
	admin.salgo();
}

process admin{
	cola c;
	int idAux;
	boolean libre = true;
	
	do persona[*]?llegue(idAux) ->
		if(libre) {
			libre = false;
			persona[idAux]!paso();
		}
		else c.push(idAux);
	[] persona[*]salgo ->
		if(not c.isEmpty()) persona[c.pop()]!paso();
		else libre = true; 
	od
}
