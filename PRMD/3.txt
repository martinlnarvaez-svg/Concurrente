Resolver el siguiente problema con PMA. En un negocio de cobros digitales hay P personas que
deben pasar por la única caja de cobros para realizar el pago de sus boletas. Las personas son
atendidas de acuerdo con el orden de llegada, teniendo prioridad aquellos que deben pagar menos
de 5 boletas de los que pagan más. Adicionalmente, las personas embarazadas tienen prioridad sobre
los dos casos anteriores. Las personas entregan sus boletas al cajero y el dinero de pago; el cajero les
devuelve el vuelto y los recibos de pago.

P personas, 1 caja de cobros
Las personas son atendidas de acuerdo al orden de llegada respetando el sig orden de prioridades:
    persona comun -> persona que paga menos de 5 boletas -> embarazadas
Las personas esperan que se les devuelva el vuelto y el recibo

chan embarazada(texto,int,int);
chan pocasBoletas(texto,int,int);
chan normal(texto,int,int);
chan recibir[P-1](int,texto);

Process persona[id: 0..P-1]{
    texto recibo,tipo,boletas; // SE CONOCE
    int dinero,cantBoletas // SE CONOCE
    
    if(tipo = embarazada){
        send embarazada(boletas,dinero,id);
    }
    else if(cantBoletas < 5){
        send pocasBoletas(boletas,dinero,id);
    }
    else{
        send normal(boletas,dinero,id);
    }
    receive recibir[id](dinero,recibo);
}

Process caja{
    texto boletas,recibo;
    int dinero,vuelto,id;
    
    do receive embarazada(boletas,dinero,id) ->
        PagarBoletas(boletas,dinero,vuelto,recibo);
        send recibir[id](vuelto,recibo);
    [] if(empty(embarazada)); receive pocasBoletas(boletas, dinero, id) ->
        PagarBoletas(boletas,dinero,vuelto,recibo);
        send recibir[id](vuelto,recibo);
    [] if(empty(embarazada) and empty(pocasBoletas)); receive normal(boletas, dinero, id) ->
        PagarBoletas(boletas,dinero,vuelto,recibo);
        send recibir[id](vuelto,recibo);
}
