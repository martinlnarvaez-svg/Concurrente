Resolver el siguiente problema. En un negocio de cobros digitales hay P personas que deben pasar
por la única caja de cobros para realizar el pago de sus boletas. Las personas son atendidas de acuerdo
con el orden de llegada, teniendo prioridad aquellos que deben pagar menos de 5 boletas de los que
pagan más. Adicionalmente, las personas ancianas tienen prioridad sobre los dos casos anteriores.
Las personas entregan sus boletas al cajero y el dinero de pago; el cajero les devuelve el vuelto y los
recibos de pago. ADA

P personas, 1 caja
se atiene en base al orden de llegada con el siguiente orden de prioridad:
    ancianos <- -5 boletas <- resto de personas
Las personas esperan el vuelo y el recibo

task caja;

task type persona is
    entry identificar(id: IN integer);
    entry fin(r: IN recibo, d IN integer);
end persona;
Vpersonas: array(0..P-1) of persona;

task admin is
    entry llegue(t: IN string, b IN string, d IN integer, c IN integer, id IN integer);
    entry pedido(boletas OUT string, dinero OUT integer, id OUT integer);
end admin;



task body persona is
    tipo, boletas, recibo: String;
    cantBoletas, dinero, id: integer;
begin
    accept identificar(id);
    admin.llegue(tipo,boletas,dinero,cantBoletas,id);
    accept fin(recibo, dinero);
end persona;

task body admin is
    tipo, boletas: String;
    dinero, id: integer;
    colaAncianos: cola;
    colaPocasBoletas: cola;
    colaRestante: cola;
begin
    for i = 0..P-1 loop
        Vpersonas(i).identificar(i);
    end loop;
    for i = 1..P*2
        select
            accept llegue(tipo: IN string, boletas IN string, dinero IN integer, cantBoletas IN integer,
            id IN integer);
                if(tipo = "anciano" || tipo = "anciana") then
                    colaAncianos.push(boletas,dinero,id);
                else if(cantBoletas < 5) then
                    colaPocasBoletas.push(boletas,dinero,id);
                else
                    colaRestante.push(boletas,dinero,id);
                end if;
            end llegue;
        or
            when(!colaAncianos.isEmpty || !colaPocasBoletas.isEmpty || !colaAncianos.isEmpty); 
            accept pedido(boletas OUT string, dinero OUT integer, id OUT integer) do
                if(!colaAncianos.isEmpty) then
                    colaAncianos.pop(boletas,dinero,id);
                else if(!colaPocasBoletas.isEmpty) then
                    colaPocasBoletas.pop(boletas,dinero,id);
                else
                    colaRestante.pop(boletas,dinero,id);
                end if;
            end pedido;
        end select;
    end loop;
end admin;

task body caja is
    boletas, recibo: String;
    dinero, id, vuelto: integer;
begin
    for i = 1..P
        caja.pedido(boletas,dinero,id);
        cobrarBoletas(boletas,dinero,vuelto,recibo);
        Vpersonas(id).fin(recibo,dinero);
    end loop;
end caja;
