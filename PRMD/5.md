Resolver el siguiente problema. En un negocio de cobros digitales hay P personas que deben pasar por la única caja de cobros para realizar el pago de sus boletas. Las personas son atendidas de acuerdo con el orden de llegada, teniendo prioridad aquellos que deben pagar menos de 5 boletas de los que pagan más. Adicionalmente, las personas ancianas tienen prioridad sobre los dos casos anteriores. Las personas entregan sus boletas al cajero y el dinero de pago; el cajero les devuelve el vuelto y los recibos de pago.
~~~

task negocio is
	entry ancianos(boletas: IN string, dinero: IN OUT integer, recibo: OUT string);
	entry pocasBoletas(boletas: IN string, dinero: IN OUT integer, recibo: OUT string);
	entry muchasBoletas(boletas: IN string, dinero: IN OUT integer, recibo: OUT string);
end negocio

task type persona;
Vpersona: array(0..P-1);

task body negocio is
	for i=0; i<P; i++ loop
		SELECT
			accept ancianos(boletas: IN string, dinero: IN OUT integer, recibo: OUT string) do
				atender(boletas,dinero,recibo);
			end aciano;
		OR
			when(ancianos'count = 0) -> accept pocasBoletas(boletas: IN string, dinero: IN OUT integer, recibo: OUT string) do
				atender(boletas,dinero,recibo);
			end pocasBoletas;
		OR
			when(ancianos'count = 0 and pocasBoletas'count = 0) -> accept muchasBoletas(boletas: IN string, dinero: IN OUT integer, recibo: OUT string) do
				atender(boletas,dinero,recibo);
			end muchasBoletas;
		end SELECT
	end loop
end negocio;

task body persona is
	tipo: string := ...;
	cantBoletas: integer:= ...;
	boletas: string := ...;
	recibo: string;
	
	if(tipo == anciano) then
		negocio.ancianos(boletas,dinero,recibo);
	else if(cantBoletas < 5) then
		negocio.pocasBoletas(boletas,dinero,recibo);
	else
		negocio.muchasBoletas(boletas,dinero,recibo);
	end if;
	
end persona;

~~~
