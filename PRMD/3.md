Resolver el siguiente problema con PMA. En un negocio de cobros digitales hay P personas que deben pasar por la única caja de cobros para realizar el pago de sus boletas. Las personas son atendidas de acuerdo con el orden de llegada, teniendo prioridad aquellos que deben pagar menos de 5 boletas de los que pagan más. Adicionalmente, las personas embarazadas tienen prioridad sobre los dos casos anteriores. Las personas entregan sus boletas al cajero y el dinero de pago; el cajero les devuelve el vuelto y los recibos de pago.
~~~
chan embarazada(id,boleta,dinero);
chan pocasBoletas(id,boleta,dinero);
chan muchasBoletas(id,boleta,dinero);
chan devolucion[P](recibo,vuelto);
chan hayPedido(boolean);


process persona[id: 0..P-1]{
	int tipo = ..;
	int cantidadBoletas = ..;
	boleta b;
	int dinero = ..;
	
	if(tipo == embarazada) send embarazada(id,boleta,dinero);
	else if(cantidadBoletas < 5) send pocasBoletas(id,boleta,dinero);
	else send muchasBoletas(id,boleta,dinero);
	send hayPedido(true);
	receive devolucion[id](recibo,vuelto);
}

process cajero{
	int dinero;
	boleta b;
	recibo r;
	int idAux;
	boolean ok;
	
	for i=0; i<P; i++{
		receive hayPedido(ok);
			if(empty(embarazadas) and empty(pocasBoletas)); receive muchasBoletas(idAux,boleta,dinero)
			[] if(empty(embarazadas)): receive pocasBoletas(idAux,boleta,dinero);
			[] receive embarazadas(idAux,boleta,dinero);
			trabajar(boleta,dinero,recibo);
			send devolucion[idAux](recibo,vuelto);
	}
}
