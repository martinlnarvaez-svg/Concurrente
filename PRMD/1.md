En una oficina existen 100 empleados que envían documentos para imprimir en 5 impresoras compartidas. Los pedidos de impresión son procesados por orden de llegada y se asignan a la primera impresora que se encuentre libre: <br>a) Implemente un programa que permita resolver el problema anterior usando PMA. <br>b) Resuelva el mismo problema anterior pero ahora usando PMS.<br>
a)
~~~
chan envioDocumento(documento,int);
chan recibirImpresion[100](impresion);

process impresora[id: 0..4]{
	impresion i;
	documento d;
	int idAux;
	
	while(true){
		receive envioDucumento(d,idAux);
		i = imprimir(d);
		send recibirImpresion[idAux](i);
	}
}

process empleado[id: 0..99]{
	documento d;
	cola impresorasLibres;
	int idAux;
	impresion i;
	
	while(true){
		d = generarDocumento();

		receive recibirImpresion[id](i);
	}
}

process impresora[id: 0..4]{
	impresion i;
	documento d;
	int idAux;
	
	while(true){
		send admin!libre(id);
		admin?envioDucumento(idAux,d);
		i = imprimir(d);
		empleado[idAux]!recibirImpresion(i);
	}
}

process empleado[id: 0..99]{
	documento d;
	cola impresorasLibres;
	int idAux;
	impresion i;
	
	while(true){
		d = generarDocumento();
		admin!envioDocumento(id,d);
		impresora[*]?recibirImpresion(i);
	}
}

process admin{
	cola pedidos;
	cola impresorasLibres;

	while(true){

	}
