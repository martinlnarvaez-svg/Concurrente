1. Resolver con PASAJE DE MENSAJES ASINCRÓNICOS (PMA) el siguiente problema. Hay E estudiantes que 
rinden un examen final y un profesor que cuando todos los estudiantes han llegado les entrega el enunciado.
Luego el profesor va corrigiendo los exámenes y enviando la nota de acuerdo con orden en que se van 
entregando. Cada alumno realiza el examen, lo entrega y espera a que el profesor le indique la nota. 
Nota: maximizar la concurrencia; no realizar busy waiting; todos los procesos deben terminar.

	chan llegaAlumno();
	chan profresorEntregaExamen[E](String);
	chan entregarExamen(string,integer);
	chan esperaNota[E](integer);
	
	Process alumno[id: 0..E-1]{
	    String e;
	    int nota;
	    send llegaAlumno();
	    receive profresorEntregaExamen(e);
	    String examen = hacerExamen(e);
	    send entregarExamen(examen,id);
	    receive esperarNota[id](nota);
	}
	
	Process profesor{
	    String examen;
	    integer id;
	    for i = 1..E do
	        receive llegaAlumno();
	    end;
	    for i = 1..E do
	        send profresorEntregaExamen[i](e);
	    end;
	    for i = 1..E do
	        receive entregarExamen(examen,id);
	        send esperarNota[id](Corregir(examen));
	}

2. Resolver con PASAJE DE MENSAJES SINCRÓNICOS (PMS) el siguiente problema. En un consultorio hay un 
médico para atender a 15 pacientes de acuerdo con el orden de llegada. Cada paciente al llegar espera 
a que el médico lo atiende y le indique su tratamiento. 
Nota: existe la función atender() que simula que el médico está atendiendo a un paciente; todos los 
procesos deben terminar.

	Process paciente[id: 0..14]{
	    String tratamiento,problema;
	    admin!llegaPaciente(id,problema);
	    medico?PacienteSeVa(tratamiento);
	}
	
	Process medico{
	    integer id;
	    String problema,tratamiento;
	    for i = 1..15 do
	        admin!libre();
	        admin?datos(id,problema);
	        tratamiento = atender(problema);
	        paciente[id]!PacienteSeVa(tratamiento);
	    end;
	}
	
	Process admin{
	    cola c;
	    integer id;
	    integer cant = 0;
	    String problema;
	    while(cant < 15){
	        if(!c.isEmpty); medico?libre() ->
	            c.pop(id,problema);
	            medico!datos(id,problema);
	            cant++;
	        [] paciente[*]?llegaPaciente(id,problema) ->
	            c.push(id,problema);
	        fi
	    }
	}


3. Resolver con ADA el siguiente problema. En una competencia de programadores hay P participantes y UN 
coordinador. El coordinador entrega el problema a resolver a los participantes y recibe las resoluciones 
para corregir. Cada participante debe conocer el resultado de su trabajo y el orden en que entregó su 
resolución. 
Nota: maximizar la concurrencia.

	task type programador is
	    entry recibirResultados(nota IN integer, orden IN integer);
	end programador;
	Vprogramadores: array(0..P-1): programador;
	
	task coordinador is
	    entry mandarProblema(problema: OUT String);
	end coordinador;
	
	task admin is
	    entry identificar(id: OUT integer);
	    entry recibirProblema(resolucion: IN String, id: IN integer);
	    entry libre(resolucion: OUT String, id: OUT integer, orden OUT integer)
	end admin
	
	
	task body admin is
	    c: cola;
	    i: integer = 0;
	    orden: integer = 0;
	begin
	    select
	        accept identificar(i: OUT integer) do
	            i:= i + 1;
	        end identificar;
	    or
	        accept recibirProblema(resolucion: IN String, id: IN integer) do
	            c.push(resolucion,id);
	        end recibirProblema
	    or
	        (!c.isEmpty); accept libre(resolucion: OUT String, id: OUT integer, orden: OUT integer) do
	            c.pop(resolucion,id);
	            orden:= orden + 1;
	        end libre;
	end admin;
	
	task body programador is
	    id,nota,orden: integer;
	    String problema,resolucion;
	begin
	    admin.identificar(id);
	    coordinador.mandarProblema(problema);
	    resolucion = resolver(problema);
	    admin.recibirProblema(resolucion,id);
	    accept recibirResultados(nota,orden);
	end programador;
	
	task body coordinador is
	    problema,resolucion: String;
	    orden,nota,id: integer;
	begin
	    select
	        accept mandarProblema(problema: OUT String);
	    or
	        admin.libre(resolucion,id,orden);
	        nota:= corregir(resolucion);
	        Vprogramadores(id).recibirResultados(nota,orden);
	    end select;
	end coordinador;
