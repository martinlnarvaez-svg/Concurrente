puede ser usado por una persona a la vez. Resuelva con MONITORES los dos casos siguientes:
a. El acceso es por orden de llegada
b. El acceso al mirador es por orden de llegada, pero dando prioridad a los mayores de 60 años


a)
process Perosna(id:0..N-1){
    Gestion.llega();
    //PASANDO

}

    int enEspera = 0;
    boolean libre = true;

    procedure llega(){
        if(!libre){
            enEspera++;
            wait(cola);
        }

            enEspera--;
            signal(cola);
        }

    }
